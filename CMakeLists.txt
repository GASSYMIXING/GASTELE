cmake_minimum_required(VERSION 3.24)
project(GASTELE VERSION 1.2 LANGUAGES CXX RC)
if(MSVC AND DEFINED GASTELE_MSVC_INCLUDE_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${GASTELE_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
if(NOT VST3_SDK_ROOT)
  message(FATAL_ERROR "Set VST3_SDK_ROOT to a VST3 SDK 3.8.0 checkout with submodules")
endif()
add_subdirectory("${VST3_SDK_ROOT}" sdk-build)
list(APPEND CMAKE_MODULE_PATH "${VST3_SDK_ROOT}/cmake/modules")
include(SMTG_Global)
set(SDK_ROOT "${VST3_SDK_ROOT}")
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
if(MSVC AND TARGET moduleinfotool)
  target_compile_options(moduleinfotool PRIVATE /Od)
endif()
smtg_add_vst3plugin(GASTELE src/Plugin.cpp src/Editor.cpp src/panel.rc)
target_include_directories(GASTELE PRIVATE "${VST3_SDK_ROOT}" src)
target_compile_definitions(GASTELE PRIVATE NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(GASTELE PRIVATE sdk gdiplus shlwapi)
add_executable(gastele_dsp_tests tests/DspTests.cpp)
target_include_directories(gastele_dsp_tests PRIVATE src)
enable_testing()
add_test(NAME DSP COMMAND gastele_dsp_tests)
add_executable(gastele_panel_render tests/PanelRender.cpp)
target_include_directories(gastele_panel_render PRIVATE src)
target_compile_definitions(gastele_panel_render PRIVATE NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(gastele_panel_render PRIVATE gdiplus shlwapi)
add_executable(gastele_integration_tests tests/IntegrationTests.cpp src/Plugin.cpp src/Editor.cpp src/panel.rc
  "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp"
  "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/parameterchanges.cpp")
target_include_directories(gastele_integration_tests PRIVATE src "${VST3_SDK_ROOT}")
target_compile_definitions(gastele_integration_tests PRIVATE NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(gastele_integration_tests PRIVATE sdk gdiplus shlwapi)
add_test(NAME Integration COMMAND gastele_integration_tests)
add_executable(gastele_realphone_tests tests/RealphoneTests.cpp)
target_include_directories(gastele_realphone_tests PRIVATE src tests)
add_test(NAME Realphone COMMAND gastele_realphone_tests)
